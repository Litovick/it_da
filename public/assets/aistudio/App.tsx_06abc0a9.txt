  );

function speak(text: string) {
  if (VOICE_DEMO_PAUSED) return;
  if (!("speechSynthesis" in window)) return;
  window.speechSynthesis.cancel();
  const utterance = new SpeechSynthesisUtterance(text);
  utterance.lang = "ko-KR";
  utterance.rate = 0.9;
  utterance.pitch = 1;
  window.speechSynthesis.speak(utterance);
}

      if (command.includes("얼마나") || command.includes("몇정거장")) {
        const message =
          stepIndex === 2
            ? "5번 버스는 지금 3정류장 뒤에 있어요. 약 6분 남았습니다."
            : stepIndex === 3
              ? `춘천시청까지 ${remainingStops}정거장 남았어요. 내릴 때 제가 먼저 알려드릴게요.`
              : `${activeStep.detail}`;
        setAssistantMessage(message);
        speak(message);
        return;
      }

      if (stepIndex === 0 && command.includes("나왔")) {
        advanceTo(1, "잘하셨어요. 이제 바로 앞 정류장으로 천천히 걸어가세요.");
        return;
      }

      if (stepIndex === 1 && command.includes("도착")) {
        advanceTo(2, "네, 정확히 도착하셨어요. 여기서 5번 버스를 기다리면 됩니다.");
        return;
      }

      if (stepIndex === 2 && command.includes("탔")) {
        setRemainingStops(4);
        advanceTo(3, "잘 타셨어요. 이제 휴대폰은 주머니에 넣어 두셔도 됩니다.");
        return;
      }

      if (stepIndex === 3 && (command.includes("내렸") || (remainingStops === 0 && command.includes("도착")))) {
        advanceTo(4, "잘 내리셨어요. 이제 오른쪽으로 천천히 걸어가세요.");
        return;
      }

      if (stepIndex === 4 && command.includes("도착")) {
        advanceTo(5, "춘천시청에 안전하게 도착했어요. 정말 잘하셨습니다.");
        return;
      }

      setAssistantMessage("말씀을 정확히 듣지 못했어요. 천천히 다시 말씀해 주세요.");
      speak("말씀을 정확히 듣지 못했어요. 천천히 다시 말씀해 주세요.");
    },
    [activeStep.detail, advanceTo, experienceMode, journeyStep.detail, journeyStep.title, remainingStops, stepIndex],
  );

  useEffect(() => {
    if (experienceMode !== "journey" || stepIndex !== 3) {
      if (rideTimerRef.current) window.clearInterval(rideTimerRef.current);
      return;
    }

    rideTimerRef.current = window.setInterval(() => {
      setRemainingStops((current) => Math.max(0, current - 1));
    }, 10000);

    return () => {
      if (rideTimerRef.current) window.clearInterval(rideTimerRef.current);
    };
  }, [experienceMode, stepIndex]);

  useEffect(() => {
    if (experienceMode !== "journey" || stepIndex !== 3) return;

    if (remainingStops === 2) {
      navigator.vibrate?.([250, 120, 250]);
      setAssistantMessage("두 정거장 뒤에 내리시면 됩니다. 아직 앉아 계셔도 괜찮아요.");
      speak("두 정거장 뒤에 내리세요. 아직 앉아 계셔도 괜찮아요. 미리 준비해 주세요.");
    }

    if (remainingStops === 1) {
      navigator.vibrate?.([400, 160, 400, 160, 400]);
      setAssistantMessage("다음 정류장에서 내리세요. 가방과 교통카드를 챙겨 주세요.");
      speak("다음 정류장에서 내리세요. 가방과 교통카드를 챙기고 손잡이를 잡아 주세요.");
    }

    if (remainingStops === 0) {
      navigator.vibrate?.([600, 180, 600, 180, 600]);
      setAssistantMessage("이번 정류장이 춘천시청입니다. 손잡이를 잡고 천천히 내리세요.");
      speak("이번 정류장에서 내리세요. 춘천시청 정류장입니다. 손잡이를 잡고 천천히 일어나세요.");
    }
  }, [experienceMode, remainingStops, stepIndex]);

  useEffect(() => {
    const timer = window.setTimeout(() => {
      speak(`${visibleStep.title} ${visibleStep.detail}`);
    }, 450);
    return () => window.clearTimeout(timer);
  }, [experienceMode, stepIndex]);

  useEffect(() => {
    if (experienceMode === "journey" && stepIndex === 5 && guardianSharing) {

      return () => window.clearTimeout(timer);

    return () => destinationRecognitionRef.current?.stop();
  }, []);

  const startDestinationVoiceSearch = () => {
    if (isDestinationListening) {
      destinationRecognitionRef.current?.stop();
      return;
    }

    const recognitionConstructor = (
      window as typeof window & {
        SpeechRecognition?: new () => SpeechRecognitionLike;
        webkitSpeechRecognition?: new () => SpeechRecognitionLike;
      }
    ).SpeechRecognition ?? (
      window as typeof window & {
        webkitSpeechRecognition?: new () => SpeechRecognitionLike;
      }
    ).webkitSpeechRecognition;

    if (!recognitionConstructor) {
      setAssistantMessage("이 브라우저에서는 음성 검색을 지원하지 않아요. 검색창에 목적지를 입력해 주세요.");
      destinationInputRef.current?.focus();
      return;
    }

    const recognition = new recognitionConstructor();
    recognition.lang = "ko-KR";
    recognition.continuous = false;
    recognition.interimResults = false;
    recognition.onstart = () => {
      setIsDestinationListening(true);
      setAssistantMessage("듣고 있어요. 가고 싶은 곳을 말씀해 주세요.");
    };
    recognition.onresult = (event) => {
      const result = event.results[event.results.length - 1][0].transcript.trim();
      setTypedDestination(result);
      setTranscript(result);
      handleCommand(result);
    };
    recognition.onerror = () => {
      setIsDestinationListening(false);
      setAssistantMessage("음성을 듣지 못했어요. 마이크를 다시 누르거나 직접 입력해 주세요.");
    };
    recognition.onend = () => setIsDestinationListening(false);
    destinationRecognitionRef.current = recognition;

    try {
      recognition.start();
    } catch {
      setIsDestinationListening(false);
      setAssistantMessage("마이크를 시작하지 못했어요. 잠시 후 다시 눌러 주세요.");

      destinationInputRef.current?.focus();
      return;
    }
    handleCommand(destination);
  };

    setTypedDestination("춘천시청");
    handleCommand("춘천시청 가려고");

    setShowHelpPanel(false);

    setTranscript("");

  };

  const openCurrentGuide = () => {
    setShowGuardianPanel(false);
    setShowHelpPanel(false);
    if (experienceMode !== "journey") {
      setAssistantMessage("먼저 홈에서 목적지를 정하면 이동 안내가 시작됩니다.");

    }
  };

  const askCurrentSituation = () => {
    setShowHelpPanel(false);
    const message =
      experienceMode === "journey"
        ? `현재 ${visibleStep.place} 단계입니다. ${visibleStep.title}`
        : "아직 이동을 시작하지 않았어요. 홈에서 목적지를 먼저 정해 주세요.";

  };

  const requestHelp = (command: string) => {
    setShowHelpPanel(false);
    if (experienceMode !== "journey") {
      setAssistantMessage(
        command.includes("도착하기")
          ? "이동을 시작하면 목적지 도착 직전에 진동과 알림으로 알려드릴게요."

    }
    handleCommand(command);
  };

  return (
    <main ref={appShellRef} className="app-shell">
      <div className="ambient ambient-one" />
      <div className="ambient ambient-two" />

      <div className="device-status" aria-label="기기 상태">
        <span>9:41</span>
        <div aria-hidden="true">
          <i className="signal-icon"><b /><b /><b /><b /></i>
          <i className="wifi-icon" />
          <i className="battery-icon"><b /></i>
        </div>
      </div>

      <header className={`app-header ${experienceMode === "setup" ? "is-home" : ""}`}>
        {experienceMode !== "setup" && <div className="home-user" aria-label="현재 안내 상태">
          <span className="home-avatar">김</span>
          <div>
            <p>안전한 이동을 함께하고 있어요</p>
            <strong>{visibleStep.phase}</strong>
          </div>
        </div>}

        <button
          type="button"
          className="guardian-status"
          onClick={() => setShowGuardianPanel(true)}
          aria-label="설정 열기"
        >
          <Icon name="gear" className="h-5 w-5" />
          <span className="guardian-status-label">설정</span>
        </button>
      </header>

      {experienceMode !== "setup" && <section className="route-summary" aria-label="현재 이동 경로">
        <div>
          <p className="route-label">
            {experienceMode === "setup" ? "안녕하세요, 김상용님" : experienceMode === "confirm" ? "말씀하신 목적지" : "오늘의 목적지"}
          </p>
          {experienceMode === "setup" ? (
            <p className="route-name"><strong>오늘도 안전한 이동을 함께할게요.</strong></p>
          ) : experienceMode === "confirm" ? (
            <p className="route-name"><strong>춘천시청</strong></p>
          ) : (
            <p className="route-name"><span>춘천역</span><span className="route-arrow">→</span><strong>춘천시청</strong></p>
          )}
          {experienceMode === "journey" && stepIndex >= 2 && (
            <p className="data-status">프로토타입 교통정보 · 실제 API 미연동</p>
          )}
        </div>
        {experienceMode === "journey" && (
          <div className="progress-count" aria-label={`${routeSteps.length}단계 중 ${stepIndex + 1}단계`}>
            <strong>{stepIndex + 1}</strong><span> / {routeSteps.length}</span>
          </div>
        )}
        <div className="progress-track" aria-hidden="true">
          <span style={{ width: `${progress}%` }} />
        </div>
      </section>}

      <section className={`guidance-card ${experienceMode === "setup" ? "is-home" : ""} ${experienceMode === "journey" && stepIndex === 3 && remainingStops <= 2 ? "is-alert" : ""} ${experienceMode === "journey" && stepIndex === 5 ? "is-complete" : ""}`} aria-live="polite">
        <div className="guidance-meta">
          <span className="phase-badge">{visibleStep.phase}</span>
          <span className="place-label">{visibleStep.place}</span>
        </div>

        <div className="action-visual" aria-hidden="true">
          <div className="action-ring ring-one" />
          <div className="action-ring ring-two" />
          <span className="action-icon"><Icon name={visibleStep.icon} className="h-12 w-12 sm:h-14 sm:w-14" /></span>
        </div>

        <div className="guidance-copy">
          <p className="now-label">지금 하실 일</p>
          <h1>{visibleStep.title}</h1>
          <p className="guidance-detail">{visibleStep.detail}</p>
        </div>

        {experienceMode === "setup" && (
          <div className="home-entry">
            <form className="destination-form" onSubmit={submitTypedDestination}>
              <span className="destination-search-icon"><Icon name="search" className="h-5 w-5" /></span>
              <input
                ref={destinationInputRef}
                type="search"
                value={typedDestination}
                onChange={(event) => setTypedDestination(event.target.value)}
                placeholder="목적지나 장소를 입력하세요"
                aria-label="목적지 검색어"
              />
              <button
                type="button"
                className={`destination-voice-button ${isDestinationListening ? "is-listening" : ""}`}
                onClick={startDestinationVoiceSearch}
                aria-label={isDestinationListening ? "음성 검색 중지" : "음성으로 목적지 검색"}
              >
                <Icon name="mic" className="h-5 w-5" />
              </button>
              <button type="submit">길 안내</button>
            </form>
            {isDestinationListening && (
              <p className="voice-search-status" aria-live="polite">
                <span /><strong>듣고 있어요.</strong> 가고 싶은 곳을 말씀해 주세요.
              </p>
            )}
          </div>
        )}

        {experienceMode === "journey" && stepIndex === 2 && (
          <div className="transit-status" aria-label="버스 도착 정보">
            <span className="transit-status-icon"><Icon name="bus" className="h-5 w-5" /></span>
            <div><small>5번 버스 도착까지</small><strong>약 6분</strong></div>
            <div><small>현재 위치</small><strong>3정류장 전</strong></div>
          </div>
        )}

        {experienceMode === "journey" && stepIndex === 3 && (
          <div className={`transit-status ${remainingStops <= 1 ? "is-near" : ""}`} aria-label="목적지까지 남은 정보">
            <span className="transit-status-icon"><Icon name="bell" className="h-5 w-5" /></span>
            <div><small>춘천시청까지</small><strong>{remainingStops === 0 ? "이번 정류장" : `${remainingStops}정류장`}</strong></div>
            <div><small>예상 시간</small><strong>{remainingStops === 0 ? "곧 도착" : `약 ${Math.max(2, remainingStops * 2)}분`}</strong></div>
          </div>
        )}

        <div className="spoken-prompt">
          <Icon name="volume" className="h-5 w-5 shrink-0" />
          <p>{visibleStep.prompt}</p>
        </div>

        {experienceMode === "confirm" && (
          <div className="confirm-actions">
            <button type="button" className="primary-action" onClick={() => handleCommand("응, 맞아")}>맞아요</button>
            <button type="button" className="secondary-action" onClick={() => handleCommand("아니, 다른 곳")}>목적지 바꾸기</button>
          </div>
        )}

        {experienceMode === "journey" && stepIndex === 5 && guardianSharing && (
          <div className="arrival-share">
            <span className="arrival-share-icon"><Icon name="shield" className="h-5 w-5" /></span>
            <div>
              <strong>보호자 안심 문자 전송 완료</strong>
              <p>“춘천시청에 안전하게 도착했습니다.”</p>
            </div>

      </section>

      {experienceMode === "setup" && (
        <>
          <button type="button" className="home-feature-card" onClick={chooseChuncheonCityHall}>
            <span className="feature-icon"><Icon name="route" className="h-6 w-6" /></span>
            <span className="feature-copy"><small>최근 이용한 안내</small><strong>춘천시청으로 바로 출발</strong><p>춘천역 출발 · 5번 버스 · 약 24분</p></span>
            <span className="feature-arrow">→</span>
          </button>

              <span>한 번 눌러 바로 안내</span>
            </div>
            <div className="favorite-grid">
              <button type="button" className="favorite-card primary" onClick={chooseChuncheonCityHall}>
                <span className="favorite-icon"><Icon name="pin" className="h-6 w-6" /></span>
                <span className="favorite-text">
                  <small>자주 방문</small>

                <span className="favorite-icon hospital"><Icon name="shield" className="h-6 w-6" /></span>

              <span>어제</span>
            </div>
            <div className="mini-route-line">
              <span className="route-node start"><i /></span>
              <div><small>출발</small><strong>춘천역 1번 출구</strong></div>
              <span className="route-bus"><Icon name="bus" className="h-4 w-4" /> 5번</span>
              <div><small>도착</small><strong>춘천시청</strong></div>
              <span className="route-node end"><i /></span>
            </div>
            <button type="button" onClick={chooseChuncheonCityHall}>이 경로로 다시 안내</button>
          </section>
        </>
      )}

      <section className="assistant-feedback" aria-live="polite">
        <span className="assistant-avatar">잇</span>
        <p>{assistantMessage}</p>
      </section>

      {experienceMode !== "setup" && <section className="voice-dock is-paused" aria-label="음성 대화 상태">
        <div className="voice-status">
          <span className="voice-orb is-paused">
            <Icon name="mic" className="h-6 w-6" />
            <i />
          </span>
          <div className="min-w-0">
            <p className="voice-label">{voiceLabel}</p>
            <p className="voice-transcript">{transcript || "아래 체험 문장을 눌러 흐름을 확인해 주세요."}</p>
          </div>
          <div className="sound-bars" aria-hidden="true">
            <i /><i /><i /><i />
          </div>
        </div>

        <div className="demo-commands">
          <p>프로토타입 체험용 · 마이크가 안 될 때 문장을 눌러보세요</p>
          <div>
            {visibleStep.demo.map((command) => (
              <button type="button" key={command} onClick={() => handleCommand(command)}>
                “{command}”
              </button>
            ))}
          </div>
        </div>
      </section>}

      {experienceMode !== "setup" && <p className="safety-note">
        <Icon name="bell" className="h-4 w-4" />
        “도착하기 직전에 말해줘”라고 하면 내릴 때 다시 알려드립니다.
      </p>}

      <nav className="bottom-nav" aria-label="주요 메뉴">
        <button type="button" className={experienceMode !== "journey" && !showGuardianPanel && !showHelpPanel ? "is-active" : ""} onClick={goHome}>
          <Icon name="home" className="h-5 w-5" />
          <span>홈</span>
        </button>

          <Icon name="route" className="h-5 w-5" />

          <Icon name="help" className="h-5 w-5" />

        <div className="modal-backdrop" role="presentation" onClick={() => setShowGuardianPanel(false)}>
          <section className="guardian-panel" role="dialog" aria-modal="true" aria-labelledby="guardian-title" onClick={(event) => event.stopPropagation()}>
            <span className="panel-icon"><Icon name="gear" className="h-7 w-7" /></span>
            <p className="panel-kicker">APP SETTINGS</p>
            <h2 id="guardian-title">설정</h2>
            <p className="panel-description">이동 중 사용할 알림과 보호자 안심 공유를 관리합니다.</p>
            <div className="settings-profile">
              <span>김</span>
              <p><strong>김상용</strong><small>71세 · 춘천시 후평동</small></p>
            </div>

              <div><span className="setting-mini-icon"><Icon name="volume" className="h-4 w-4" /></span><p><strong>음성 안내</strong><small>현재 잠시 멈춤</small></p><b>꺼짐</b></div>
              <div><span className="setting-mini-icon"><Icon name="bell" className="h-4 w-4" /></span><p><strong>진동 알림</strong><small>두 정거장 전과 도착 직전 알림</small></p><b className="is-on">켜짐</b></div>
            </div>
